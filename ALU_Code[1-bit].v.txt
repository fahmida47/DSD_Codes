module alu_1bit(
    input A,
    input B,
    input Cin,
    input [2:0] ALU_Sel,

    output reg ALU_Out,
    output reg CarryOut
);

always @(*) begin

    CarryOut = 0;

    case(ALU_Sel)

        3'b000: ALU_Out = A ^ B;                 // XOR

        3'b001: ALU_Out = ~(A ^ B);              // XNOR

        3'b010: ALU_Out = ~(A & B);              // NAND

        3'b011: ALU_Out = ~A;                    // NOT A

        3'b100: ALU_Out = A & (~B);              // A AND NOT B

        3'b101: ALU_Out = (~(A ^ B)) & (A | ~B); // XNOR AND (A OR NOT B)

        3'b110:
            {CarryOut, ALU_Out} = A + B + Cin;   // Addition

        3'b111:
            {CarryOut, ALU_Out} = A + (~B) + Cin;// Subtraction

    endcase

end

endmodule
